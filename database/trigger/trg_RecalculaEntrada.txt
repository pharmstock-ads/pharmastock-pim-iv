CREATE OR ALTER TRIGGER trg_RecalculaEntrada
ON EntradaItem
AFTER INSERT, UPDATE, DELETE
AS
BEGIN
    set nocount on;
    update e
    set valor_total = COALESCE((
          select sum(ee.preco_unitario * ee.quantidade)
            from entradaitem ee
            where ee.id_entrada = e.id_entrada
        ), 0)
    from entrada e
    where e.id_entrada in (
        select 
            id_entrada
        from deleted
        union
        select 
            id_entrada
        from inserted)
END;
GO
