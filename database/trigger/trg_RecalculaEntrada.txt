CREATE OR ALTER TRIGGER trg_RecalculaEntrada
ON EntradaItem
AFTER INSERT, UPDATE, DELETE
AS
BEGIN
    set nocount on;

    declare @vIdEntrada int;

    select @vIdEntrada = (
        select id_entrada
        from deleted
        union
        select id_entrada
        from inserted
    );

    update entrada
    set valor_total = COALESCE((
        select sum(e.preco_unitario * e.quantidade)
        from entradaitem e
        where e.id_entrada = @vIdEntrada
    ), 0)
    where id_entrada = @vIdEntrada;
END;
GO
