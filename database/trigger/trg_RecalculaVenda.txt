CREATE OR ALTER TRIGGER trg_RecalculaVenda
ON VendaItem
AFTER INSERT, UPDATE, DELETE
AS
BEGIN
    set nocount on;
    update e
    set valor_total = COALESCE((
          select sum(ee.preco_unitario * ee.quantidade)
            from vendaitem ee
            where ee.id_venda = e.id_venda
        ), 0)
    from venda e
    where e.id_venda in (
        select 
            id_venda
        from deleted
        union
        select 
            id_venda
        from inserted)
END;
GO
